public class GetTrDeviceBackhaulListService

ApWifiQualityDO apWifiQualityDO = apWifiQualityDAO.findLatestApWifiQuality(networkId, mac,
    System.currentTimeMillis());
if (apWifiQualityDO != null) {
    for (TrBackhaulInfoDTO trBackhaulInfoDTO : trBackhaulInfoDTOList) {
        BackhaulTypeEnum backhaulType = BackhaulTypeEnum.value(trBackhaulInfoDTO.getType());
        if (backhaulType == BackhaulTypeEnum.TYPE_2G4) {
            trBackhaulInfoDTO.setAvlBandwidth(apWifiQualityDO.getData().getAvlBandwidth());
            trBackhaulInfoDTO.setAur(apWifiQualityDO.getData().getBackhaulAur2g4());
        }
        if (backhaulType == BackhaulTypeEnum.TYPE_5G1 || backhaulType == BackhaulTypeEnum.TYPE_5G2) {
            trBackhaulInfoDTO.setAvlBandwidth(apWifiQualityDO.getData().getAvlBandwidth());
            trBackhaulInfoDTO.setAur(apWifiQualityDO.getData().getBackhaulAur5g());
        }
        if (backhaulType == BackhaulTypeEnum.TYPE_6G) {
            trBackhaulInfoDTO.setAvlBandwidth(apWifiQualityDO.getData().getAvlBandwidth());
            trBackhaulInfoDTO.setAur(apWifiQualityDO.getData().getBackhaulAur6g());
        }
    }
}
